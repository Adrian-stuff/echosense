cmake_minimum_required(VERSION 3.10)
project(echosense_kinect2 CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(echosense-kinect2 kinect2_bridge.cpp)
find_package(Threads REQUIRED)
target_link_libraries(echosense-kinect2 Threads::Threads)

# libfreenect2 installs a CMake package config (freenect2Config.cmake)
find_package(freenect2 QUIET)
if(freenect2_FOUND)
  message(STATUS "libfreenect2 found: ${freenect2_INCLUDE_DIRS}")
  target_compile_definitions(echosense-kinect2 PRIVATE ECHOSENSE_WITH_FREENECT2)
  target_include_directories(echosense-kinect2 PRIVATE ${freenect2_INCLUDE_DIRS})
  target_link_libraries(echosense-kinect2 ${freenect2_LIBRARIES})
else()
  message(WARNING "libfreenect2 not found: building echosense-kinect2 with --fake mode only")
endif()

install(TARGETS echosense-kinect2 DESTINATION bin)
